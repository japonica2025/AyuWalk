# Ayu Walk Foundation Slice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first executable Ayu Walk / 织步记 iOS foundation slice: project scaffold, core data contracts, sample trip generation, local app shell, and a minimal planning-to-journal preview loop.

**Architecture:** Start local-first with a testable Swift package for core models and engines, then connect it to a SwiftUI iOS app shell. The first slice avoids real AI, MapKit editing, persistence, and export complexity; it proves the shared data model and screen flow that later AI, MapKit, SwiftData, MiniMax, and export features will plug into.

**Tech Stack:** Swift 5.9+, Swift Package Manager, SwiftUI, XCTest, iOS app target, future-ready interfaces for SwiftData, MapKit, AIService, JournalEngine, ExportService, and PlanningScriptEngine.

---

## Design Read

Reading this as: native iOS consumer travel/productivity app for independent travelers, with a minimal journal language, leaning toward SwiftUI-native interaction, clear utility surfaces, and restrained handcrafted travel-journal accents.

For visual decisions in this slice:

- `DESIGN_VARIANCE: 5`
- `MOTION_INTENSITY: 3`
- `VISUAL_DENSITY: 4`

Use taste-skill as a visual calibration aid for future share cards, journal templates, spacing, and page hierarchy. Do not apply its web-specific React/Tailwind guidance to native iOS implementation.

## Scope Split

The MVP spec contains multiple independent subsystems. This plan implements only the first vertical slice.

Covered in this plan:

- Repository structure for an iOS app plus testable Swift package.
- Core data model names and relationships.
- Sample trip fixture.
- Mock planning engine that returns structured itinerary data.
- Mock journal engine that generates cover, overview, and daily pages.
- SwiftUI app shell with tabs for Plan and Journal.
- Basic visual direction tokens.

Explicitly deferred to later plans:

- Real MiniMax integration.
- Real text model integration.
- MapKit route lines and numbered markers.
- SwiftData persistence.
- Page-turning animation.
- Sticker placement editing.
- Budget and AA calculator UI.
- Packing list UI.
- Image/PDF/Markdown export.
- Reminders and lock-screen widgets.
- iCloud sync and account system.

```swift
import Foundation

public struct Trip: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var title: String
    public var englishProductName: String
    public var destination: String
    public var purpose: [TravelPurpose]
    public var duration: TripDuration
    public var days: [TripDay]
    public var participants: [Participant]
    public var importedSources: [ImportedSource]
    public var budgetPlan: BudgetPlan?
    public var packingList: PackingList?
    public var journalPages: [JournalPage]
    public var planningScripts: [PlanningScript]

    public init(
        id: UUID,
        title: String,
        englishProductName: String,
        destination: String,
        purpose: [TravelPurpose],
        duration: TripDuration,
        days: [TripDay],
        participants: [Participant],
        importedSources: [ImportedSource],
        budgetPlan: BudgetPlan?,
        packingList: PackingList?,
        journalPages: [JournalPage],
        planningScripts: [PlanningScript]
    ) {
        self.id = id
        self.title = title
        self.englishProductName = englishProductName
        self.destination = destination
        self.purpose = purpose
        self.duration = duration
        self.days = days
        self.participants = participants
        self.importedSources = importedSources
        self.budgetPlan = budgetPlan
        self.packingList = packingList
        self.journalPages = journalPages
        self.planningScripts = planningScripts
    }
}

public enum TravelPurpose: String, CaseIterable, Codable, Equatable, Sendable {
    case concert
    case honeymoon
    case family
    case friends
    case cityWalk
    case shopping
    case food
}

public enum TripDuration: Equatable, Sendable {
    case dayCount(Int)
    case dateRange(start: Date, end: Date)
}

public struct TripDay: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var dayNumber: Int
    public var dateLabel: String
    public var title: String
    public var activities: [Activity]

    public init(id: UUID, dayNumber: Int, dateLabel: String, title: String, activities: [Activity]) {
        self.id = id
        self.dayNumber = dayNumber
        self.dateLabel = dateLabel
        self.title = title
        self.activities = activities
    }
}

public struct Activity: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var title: String
    public var kind: ActivityKind
    public var place: Place?
    public var startTime: String?
    public var endTime: String?
    public var notes: String
    public var estimatedCost: Decimal?
    public var routeOrder: Int?
    public var reminder: Reminder?

    public init(
        id: UUID,
        title: String,
        kind: ActivityKind,
        place: Place?,
        startTime: String?,
        endTime: String?,
        notes: String,
        estimatedCost: Decimal?,
        routeOrder: Int?,
        reminder: Reminder?
    ) {
        self.id = id
        self.title = title
        self.kind = kind
        self.place = place
        self.startTime = startTime
        self.endTime = endTime
        self.notes = notes
        self.estimatedCost = estimatedCost
        self.routeOrder = routeOrder
        self.reminder = reminder
    }
}

public enum ActivityKind: String, Codable, Equatable, Sendable {
    case sight
    case meal
    case hotel
    case transport
    case shopping
    case concert
    case freeTime
    case note
}

public struct Place: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var name: String
    public var address: String?
    public var latitude: Double?
    public var longitude: Double?
    public var providerIDs: [MapProviderKind: String]

    public init(
        id: UUID,
        name: String,
        address: String?,
        latitude: Double?,
        longitude: Double?,
        providerIDs: [MapProviderKind: String]
    ) {
        self.id = id
        self.name = name
        self.address = address
        self.latitude = latitude
        self.longitude = longitude
        self.providerIDs = providerIDs
    }
}

public enum MapProviderKind: String, Codable, Hashable, Sendable {
    case mapKit
    case google
    case mapbox
}

public struct Participant: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var name: String

    public init(id: UUID, name: String) {
        self.id = id
        self.name = name
    }
}

public struct ImportedSource: Identifiable, Equatable, Sendable {
    public var id: UUID
    public var kind: ImportedSourceKind
    public var title: String
    public var extractedText: String

    public init(id: UUID, kind: ImportedSourceKind, title: String, extractedText: String) {
        self.id = id
        self.kind = kind
        self.title = title
        self.extractedText = extractedText

public enum ImportedSourceKind: String, Codable, Equatable, Sendable {

        self.total = total

        self.id = id

        self.summary = summary
    }
}
```

```swift
                    JournalBlock(id: UUID(), kind: .timeline, title: "时间轴", body: timelineSummary(for: day), isDefaultSelected: false),
                    JournalBlock(id: UUID(), kind: .mapSnapshot, title: "地图", body: "地图快照将在后续 MapKit 计划中接入", isDefaultSelected: false),
                    JournalBlock(id: UUID(), kind: .sticker, title: "贴纸", body: "天气、交通、美食、景点、心情、酒店、购物、演唱会", isDefaultSelected: false)
                ]
            )
        }

        return [cover, overview] + dailyPages
    }

    private func routeSummary(for trip: Trip) -> String {
        trip.days
            .flatMap(\.activities)
            .compactMap { activity in
                guard let routeOrder = activity.routeOrder else { return nil }
                return "\(routeOrder). \(activity.title)"
            }
            .joined(separator: "\n")
    }

    private func budgetSummary(for trip: Trip) -> String {
        guard let budgetPlan = trip.budgetPlan else {
            return "未设置预算"
        }
        return "总预算 \(budgetPlan.total) \(budgetPlan.currencyCode)"
    }

    private func packingSummary(for trip: Trip) -> String {
        guard let packingList = trip.packingList else {
            return "未生成行李清单"
        }
        return packingList.items.map { item in
            item.isPacked ? "已完成 \(item.title)" : "待打包 \(item.title)"
        }.joined(separator: "\n")
    }

    private func timelineSummary(for day: TripDay) -> String {
        day.activities.map { activity in
            let time = [activity.startTime, activity.endTime]
                .compactMap { $0 }
                .joined(separator: "-")
            return time.isEmpty ? activity.title : "\(time) \(activity.title)"
        }.joined(separator: "\n")
    }
}
```

```swift
import AyuWalkCore
```

```swift
struct ItineraryTimelineView: View {
    let days: [TripDay]

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            ForEach(days) { day in
                daySection(day)
            }
        }
    }

    private func daySection(_ day: TripDay) -> some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("\(day.dateLabel)｜\(day.title)")
                .font(.headline)
                .foregroundStyle(AyuWalkTheme.ink)

            ForEach(day.activities) { activity in
                HStack(alignment: .top, spacing: 12) {
                    routeBadge(activity.routeOrder)

                    VStack(alignment: .leading, spacing: 4) {
                        Text(activity.title)
                            .font(.body.weight(.semibold))
                            .foregroundStyle(AyuWalkTheme.ink)

                        if let time = timeLabel(for: activity) {
                            Text(time)
                                .font(.caption)
                                .foregroundStyle(AyuWalkTheme.accent)
                        }

                        Text(activity.notes)
                            .font(.callout)
                            .foregroundStyle(AyuWalkTheme.mutedInk)
                    }
                }
            }
        }
        .padding()
        .background(AyuWalkTheme.paper)
        .clipShape(RoundedRectangle(cornerRadius: AyuWalkTheme.cardRadius))
        .overlay(
            RoundedRectangle(cornerRadius: AyuWalkTheme.cardRadius)
                .stroke(AyuWalkTheme.border)
        )
    }

    private func routeBadge(_ order: Int?) -> some View {
        Text(order.map(String.init) ?? "-")
            .font(.caption.weight(.bold))
            .foregroundStyle(.white)
            .frame(width: 26, height: 26)
            .background(AyuWalkTheme.accent)
            .clipShape(Circle())
    }

    private func timeLabel(for activity: Activity) -> String? {
        let parts = [activity.startTime, activity.endTime].compactMap { $0 }
        return parts.isEmpty ? nil : parts.joined(separator: "-")
    }
}
```
